-- =====================================================
-- QUESTION 1: Count the records where all values are NULL
-- =====================================================
-- Explanation:
-- I check all 9 columns of the invoice table with IS NULL and combine
-- the conditions with AND. This way, only rows where every column is
-- NULL are selected. COUNT(*) then returns the number of those rows.

SELECT COUNT(*)
FROM invoice
WHERE invoice_id IS NULL
    AND customer_id IS NULL
    AND invoice_date IS NULL
    AND billing_address IS NULL
    AND billing_city IS NULL
    AND billing_state IS NULL
    AND billing_country IS NULL
    AND billingpostal_code IS NULL
    AND total IS NULL;

-- Row count: 0


-- =====================================================
-- QUESTION 2: Double the total values and compare with the old values
-- =====================================================
-- Explanation:
-- I show the original total column next to a new column called
-- total_double, which is total * 2. With ORDER BY total_double DESC,
-- I sort the result from largest to smallest by the new version column.

SELECT invoice_id, total, total * 2 AS total_double
FROM invoice
ORDER BY total_double DESC;

-- Row count: 412


-- =====================================================
-- QUESTION 3: "Açık Adres" (first 3 + last 4 characters), filtered to August 2013
-- =====================================================
-- Explanation:
-- With SUBSTRING, I take the first 3 characters and the last 4 characters
-- of the billing_address column. CONCAT joins these two parts, and I name
-- the result "Açık Adres". In the WHERE clause, I limit invoice_date to
-- the range from 1 August 2013 up to (but not including) 1 September 2013,
-- so only records from August 2013 are returned.

SELECT CONCAT(
           SUBSTRING(billing_address FROM 1 FOR 3),
           SUBSTRING(billing_address FROM LENGTH(billing_address) - 3 FOR 4)
       ) AS "Açık Adres"
FROM invoice
WHERE invoice_date >= '2013-08-01'
  AND invoice_date < '2013-09-01';

-- Row count: 7
